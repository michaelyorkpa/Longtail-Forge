import assert from "node:assert/strict";
import { it } from "vitest";
import { recordsService } from "../../src/modules/__ID__/records.service.js";

it("returns the empty __TITLE__ starting-point page", async () => {
  const result = await recordsService.list({ workspace_id: "scaffold-proof" });
  assert.deepEqual(result.data, []);
  assert.equal(result.pagination.total, 0);
  assert.equal(result.pagination.has_more, false);
});
